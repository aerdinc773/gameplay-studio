BASE_DIR = Path(__file__).resolve().parents[2]

DATA_DIR = BASE_DIR / "data"

DATABASE_FILE = (
    DATA_DIR / "channel_analysis.db"
)


# ============================================================
# BAĞLANTI
# ============================================================


def get_connection():

    DATA_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )

    connection = sqlite3.connect(
        DATABASE_FILE
    )

    connection.row_factory = sqlite3.Row

    connection.execute(
        "PRAGMA foreign_keys = ON"
    )

    return connection

def utc_now():

    return datetime.now(
        timezone.utc
    ).isoformat()

def initialize_channel_database():

    connection = get_connection()

    cursor = connection.cursor()

    # ========================================================
    # KANALLAR
    # ========================================================

    cursor.execute(
        """
        CREATE TABLE IF NOT EXISTS channels (

            id INTEGER PRIMARY KEY AUTOINCREMENT,

            channel_id TEXT NOT NULL UNIQUE,

            title TEXT,

            description TEXT,

            published_at TEXT,

            country TEXT,

            subscriber_count INTEGER DEFAULT 0,

            view_count INTEGER DEFAULT 0,

            video_count INTEGER DEFAULT 0,

            source_url TEXT,

            created_at TEXT,

            updated_at TEXT,

            last_analyzed_at TEXT

        )
        """
    )

    # ========================================================
    # VİDEOLAR
    # ========================================================

    cursor.execute(
        """
        CREATE TABLE IF NOT EXISTS videos (

            id INTEGER PRIMARY KEY AUTOINCREMENT,

            video_id TEXT NOT NULL UNIQUE,

            channel_id TEXT NOT NULL,

            title TEXT,

            description TEXT,

            published_at TEXT,

            category_id TEXT,

            duration TEXT,

            duration_seconds REAL DEFAULT 0,

            is_short INTEGER DEFAULT 0,

            video_type TEXT,

            views INTEGER DEFAULT 0,

            likes INTEGER DEFAULT 0,

            comments INTEGER DEFAULT 0,

            engagement_rate REAL DEFAULT 0,

            tag_count INTEGER DEFAULT 0,

            created_at TEXT,

            updated_at TEXT,

            FOREIGN KEY (
                channel_id
            )
            REFERENCES channels (
                channel_id
            )
            ON DELETE CASCADE

        )
        """
    )

    # ========================================================
    # ETİKETLER
    # ========================================================

    cursor.execute(
        """
        CREATE TABLE IF NOT EXISTS video_tags (

            id INTEGER PRIMARY KEY AUTOINCREMENT,

            video_id TEXT NOT NULL,

            tag TEXT NOT NULL,

            created_at TEXT,

            FOREIGN KEY (
                video_id
            )
            REFERENCES videos (
                video_id
            )
            ON DELETE CASCADE

        )
        """
    )

    cursor.execute(
        """
        CREATE UNIQUE INDEX IF NOT EXISTS
        idx_video_tags_unique

        ON video_tags (
            video_id,
            tag
        )
        """
    )

    cursor.execute(
        """
        CREATE INDEX IF NOT EXISTS
        idx_videos_channel

        ON videos (
            channel_id
        )
        """
    )

    cursor.execute(
        """
        CREATE INDEX IF NOT EXISTS
        idx_videos_published

        ON videos (
            published_at
        )
        """
    )

    # ========================================================
    # ANALİZ GEÇMİŞİ
    # ========================================================

    cursor.execute(
        """
        CREATE TABLE IF NOT EXISTS analysis_runs (

            id INTEGER PRIMARY KEY AUTOINCREMENT,

            channel_id TEXT NOT NULL,

            analyzed_at TEXT NOT NULL,

            requested_video_count INTEGER DEFAULT 0,

            analyzed_video_count INTEGER DEFAULT 0,

            shorts_count INTEGER DEFAULT 0,

            long_count INTEGER DEFAULT 0,

            total_views INTEGER DEFAULT 0,

            total_likes INTEGER DEFAULT 0,

            total_comments INTEGER DEFAULT 0,

            total_tags INTEGER DEFAULT 0,

            source_url TEXT,

            FOREIGN KEY (
                channel_id
            )
            REFERENCES channels (
                channel_id
            )
            ON DELETE CASCADE

        )
        """
    )

    cursor.execute(
        """
        CREATE INDEX IF NOT EXISTS
        idx_analysis_runs_channel

        ON analysis_runs (
            channel_id
        )
        """
    )

    connection.commit()

    connection.close()

    print()
    print(
        "========================================"
    )
    print(
        "📊 EXTERNAL CHANNEL DATABASE"
    )
    print(
        "========================================"
    )
    print()
    print(
        "✅ Dış kanal analiz veritabanı hazır."
    )
    print()
    print(
        f"📁 {DATABASE_FILE}"
    )

            country,
            subscriber_count,
            view_count,
            video_count,
            source_url,

            ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?

            title = excluded.title,

                excluded.description,

                excluded.country,

        (

            channel.get(

                ""

            channel.get(
                "title",
                ""
            ),

            channel.get(
                "description",
                ""
            ),

        )

                excluded.channel_id,

            title =
                excluded.title,

            description =
                excluded.description,

            published_at =
                excluded.published_at,

            category_id =
                excluded.category_id,

                excluded.duration,

                excluded.duration_seconds,

                excluded.is_short,

                excluded.video_type,

                excluded.views,

                excluded.likes,

                excluded.comments,

            video.get(
                "published_at",

def save_analysis_run(
    channel_id,
    videos,
    source_url=None,
    requested_video_count=0,
):

    initialize_channel_database()

    connection = get_connection()

    cursor = connection.cursor()

    analyzed_count = len(
        videos
    )

    shorts_count = sum(
        1
        for video in videos
        if video.get(
            "is_short",
            False
        )
    )

    long_count = (
        analyzed_count
        - shorts_count
    )

    total_views = sum(
        video.get(
            "views",
            0
        )
        for video in videos
    )

    total_likes = sum(
        video.get(
            "likes",
            0
        )
        for video in videos
    )

    total_comments = sum(
        video.get(
            "comments",
            0
        )
        for video in videos
    )

    total_tags = sum(
        len(
            video.get(
                "tags",
                []
            )
        )
        for video in videos
    )

    cursor.execute(
        """
        INSERT INTO analysis_runs (

            channel_id,
            analyzed_at,
            requested_video_count,
            analyzed_video_count,
            shorts_count,
            long_count,
            total_views,
            total_likes,
            total_comments,
            total_tags,
            source_url

        )

        VALUES (
            ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?
        )
        """,
        (

            channel_id,

            utc_now(),

            requested_video_count,

            analyzed_count,

            shorts_count,

            long_count,

            total_views,

            total_likes,

            total_comments,

            total_tags,

            source_url,
        ),
    )

    connection.commit()

    run_id = cursor.lastrowid

    connection.close()

    return run_id

def save_channel_analysis(
    channel,
    videos,
    source_url=None,
    requested_video_count=0,
):

    channel_id = channel.get(
        "channel_id"
    )

    if not channel_id:

        raise ValueError(
            "channel_id bulunamadı."
        )

    save_channel(
        channel,
        source_url=source_url,
    )

    for video in videos:

        video["channel_id"] = (
            channel_id
        )

        save_video(
            video
        )

    run_id = save_analysis_run(
        channel_id=channel_id,
        videos=videos,
        source_url=source_url,
        requested_video_count=(
            requested_video_count
        ),
    )

    return run_id

    )

    description = description_file.read_text(

        )

    if not tags:
        raise ValueError(
            f"Hashtag bulunamadı: {hashtag_file}"
        )

    return title, description, tags

def parse_duration(duration):

    import re

    match = re.fullmatch(
        r"PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?",
        duration
    )

    if not match:
        return 0

    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    seconds = int(match.group(3) or 0)

    return (
        hours * 3600
        + minutes * 60
        + seconds
    )

    videos = [
        row_to_dict(video)

    if not videos:

            "total_videos": 0,
            "shorts_videos": 0,
            "long_videos": 0,
            "total_views": 0,
            "total_likes": 0,
            "total_comments": 0,
            "average_views": 0,
            "average_likes": 0,
            "average_comments": 0,
            "engagement_rate": 0,
            "average_duration": 0,
            "top_topics": [],
            "top_videos": [],
            "top_shorts": [],
            "top_long_videos": [],
            "performance_available": False,
        }

    # --------------------------------------------------------
    # Süreyi saniyeye çevir
    # --------------------------------------------------------

    def duration_to_seconds(value):

        if isinstance(
            value,
            (int, float)
        ):

            return float(value)

        if not value:

            return 0

        match = re.match(
            r"^PT"
            r"(?:(\d+)H)?"
            r"(?:(\d+)M)?"
            r"(?:(\d+)S)?$",
            str(value)
        )

        if not match:

            return 0

        hours = int(
            match.group(1) or 0
        )

        minutes = int(
            match.group(2) or 0
        )

        seconds = int(
            match.group(3) or 0
        )

        return (
            hours * 3600
            + minutes * 60
            + seconds
        )

    # --------------------------------------------------------
    # Videoları normalize et
    # --------------------------------------------------------

    normalized = []

    for video in videos:

        item = dict(video)

        item["duration"] = (
            duration_to_seconds(
                item.get("duration")
            )
        )

        item["views"] = int(
            item.get("views") or 0
        )

        item["likes"] = int(
            item.get("likes") or 0
        )

        item["comments"] = int(
            item.get("comments") or 0
        )

        item["topic"] = (
            item.get("topic")
            or item.get("title")
            or ""
        )

        normalized.append(
            item
        )

    # --------------------------------------------------------
    # Shorts / uzun video
    # --------------------------------------------------------

    shorts = [
        video
        for video in normalized
        if video["duration"] <= 180
    ]

    long_videos = [
        video
        for video in normalized
        if video["duration"] > 180
    ]

    # --------------------------------------------------------
    # Temel toplamlar
    # --------------------------------------------------------

    total_views = sum(
        video["views"]
        for video in normalized
    )

    total_likes = sum(
        video["likes"]
        for video in normalized
    )

    total_comments = sum(
        video["comments"]
        for video in normalized
    )

    total_count = len(
        normalized
    )

    average_views = (
        total_views / total_count
        if total_count
        else 0
    )

    average_likes = (
        total_likes / total_count
        if total_count
        else 0
    )

    average_comments = (
        total_comments / total_count
        if total_count
        else 0
    )

    durations = [
        video["duration"]
        for video in normalized
        if video["duration"] > 0
    ]

    average_duration = (
        sum(durations)
        / len(durations)
        if durations
        else 0
    )

    engagement_rate = (
        (
            total_likes
            + total_comments
        )
        / total_views
        * 100
        if total_views
        else 0
    )

    # --------------------------------------------------------
    # Konular / başlıklar
    # --------------------------------------------------------

    topic_counter = Counter(
        video["topic"]
        for video in normalized
        if video["topic"]
    )

    top_topics = [
        {
            "topic": topic,
            "count": count,
        }

        for topic, count
        in topic_counter.most_common(20)
    ]

    # --------------------------------------------------------
    # En başarılı videolar
    # --------------------------------------------------------

    sorted_videos = sorted(
        normalized,
        key=lambda video:
            video["views"],
        reverse=True,
    )

    # --------------------------------------------------------

            video["views"],
        reverse=True,
    )

    # --------------------------------------------------------
    # Uzun video özel analiz
    # --------------------------------------------------------

    long_sorted = sorted(
        long_videos,
        key=lambda video:
            video["views"],
        reverse=True,
    )

    # --------------------------------------------------------
    # SONUÇ
    # --------------------------------------------------------

    return {

def format_srt_time(seconds):

    milliseconds = int(round(seconds * 1000))

    hours = milliseconds // 3_600_000
    milliseconds %= 3_600_000

    minutes = milliseconds // 60_000
    milliseconds %= 60_000

    seconds = milliseconds // 1000
    milliseconds %= 1000

    return (
        f"{hours:02d}:"
        f"{minutes:02d}:"
        f"{seconds:02d},"
        f"{milliseconds:03d}"
    )

def create_subtitle(
    transcript_file,
    audio_file,
    output_file,
):

    audio_file = Path(audio_file)
    output_file = Path(output_file)

    if not transcript_file.exists():
        raise FileNotFoundError(
            f"Transkript bulunamadı: {transcript_file}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    text = transcript_file.read_text(
        encoding="utf-8"
    ).strip()

    if not text:
        raise ValueError(
            "Transkript boş."
        )

    duration = get_audio_duration(
        audio_file
    )

    print()
    print("=== AI ALTYAZI MOTORU ===")
    print(
        f"🎙️ Ses süresi: {duration:.2f} saniye"
    )

    print(
        "🤖 AI doğal altyazı bloklarını oluşturuyor..."
    )

    captions = create_ai_captions(
        text
    )

    print(
        f"📝 {len(captions)} altyazı bloğu oluşturuldu."
    )

    # --------------------------------------------------------
    # KARAKTER TOPLAMI
    # --------------------------------------------------------

    total_characters = sum(
        len(caption)
        for caption in captions
    )

    if total_characters <= 0:
        raise ValueError(
            "Altyazı karakter sayısı geçersiz."
        )

    # --------------------------------------------------------
    # ZAMANLAMA
    # --------------------------------------------------------

    current_time = 0.0

    srt_blocks = []

    for index, caption in enumerate(
        captions,
        start=1
    ):

        ratio = (
            len(caption)
            / total_characters
        )

        segment_duration = (
            duration * ratio
        )

        start_time = current_time

        if index == len(captions):

            end_time = duration

        else:

            end_time = (
                current_time
                + segment_duration
            )

        srt_blocks.append(
            f"{index}\n"
            f"{format_srt_time(start_time)} --> "
            f"{format_srt_time(end_time)}\n"
            f"{caption}\n"
        )

        current_time = end_time

    # --------------------------------------------------------
    # KAYDET
    # --------------------------------------------------------

    output_file.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    output_file.write_text(
        "\n".join(srt_blocks),
        encoding="utf-8"
    )

    print("=== ALTYAZI BLOKLARI ===")

        captions,
        start=1
    ):

            f"{index:02d}: {caption}"
        )

    print()
    print("✅ AI ALTYAZI OLUŞTURULDU")
    print(f"Dosya: {output_file}")

    return output_file

def is_short_duration(duration):

    if not duration:
        return False

    match = re.fullmatch(
        r"PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?",
        duration
    )

    if not match:
        return False

    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    seconds = int(match.group(3) or 0)

    total_seconds = (
        hours * 3600
        + minutes * 60
        + seconds
    )

    return total_seconds <= 180

def resolve_channel_id(
    youtube,
    channel_url,
):

    url = normalize_url(
        channel_url
    )

    parsed = urlparse(url)

    host = parsed.netloc.lower()

    if host not in (
        "youtube.com",
        "www.youtube.com",
        "m.youtube.com",
    ):

        raise ChannelAnalyzerError(
            "Geçerli bir YouTube kanal URL'si gir."
        )

    path = parsed.path.rstrip("/")

    # --------------------------------------------------------
    # /channel/UC...
    # --------------------------------------------------------

    match = re.match(
        r"^/channel/(UC[\w-]+)$",
        path,
        re.IGNORECASE,
    )

    if match:

        return match.group(1)

    # --------------------------------------------------------
    # /@handle
    # --------------------------------------------------------

    if path.startswith("/@"):

        handle = path[2:]

        response = youtube.channels().list(
            part="id",
            forHandle=handle,
        ).execute()

        items = response.get(
            "items",
            []
        )

        if items:

            return items[0]["id"]

        raise ChannelAnalyzerError(
            f"Kanal bulunamadı: @{handle}"
        )

    # --------------------------------------------------------
    # /c/kanal
    # /user/kanal
    # --------------------------------------------------------

    for prefix in (
        "/c/",
        "/user/",
    ):

        if path.startswith(prefix):

            name = path[
                len(prefix):
            ]

            response = youtube.search().list(
                part="snippet",
                q=name,
                type="channel",
                maxResults=5,
            ).execute()

            items = response.get(
                "items",
                []
            )

            if items:

                return (
                    items[0]
                    ["snippet"]
                    ["channelId"]
                )

    raise ChannelAnalyzerError(
        "Kanal ID'si bulunamadı."
    )

    short_ids = set()

    # --------------------------------------------------------
    # /shorts/VIDEO_ID
    # --------------------------------------------------------

    matches = re.findall(
        r"/shorts/"
        r"([A-Za-z0-9_-]{11})",
        html,
    )

    short_ids.update(
        matches
    )

    # --------------------------------------------------------
    # ytInitialData içindeki videoId alanları
    # --------------------------------------------------------

    matches = re.findall(
        r'"videoId"\s*:\s*'
        r'"([A-Za-z0-9_-]{11})"',
        html,
    )

    for video_id in matches:

        position = html.find(
            f'"videoId":"{video_id}"'
        )

        if position == -1:

            continue

        start = max(
            0,
            position - 1000
        )

        context = html[
            start:
            position + 300
        ]

        if (
            "/shorts/"
            in context
            or "reelShelf"
            in context
            or "Shorts"
            in context
            or "shorts"
            in context
        ):

            short_ids.add(
                video_id
            )

    return short_ids

# ============================================================


def duration_to_seconds(
    duration,
):

    if isinstance(
        duration,
        (int, float)
    ):

        return float(duration)

    if not duration:

        return 0

    match = re.match(
        r"^PT"
        r"(?:(\d+)H)?"
        r"(?:(\d+)M)?"
        r"(?:(\d+)S)?$",
        str(duration),
    )

    if not match:

        return 0

    hours = int(
        match.group(1) or 0
    )

    minutes = int(
        match.group(2) or 0
    )

    seconds = int(
        match.group(3) or 0
    )

    return (
        hours * 3600
        + minutes * 60
        + seconds
    )

def is_retryable_error(error):
    """
    Hatanın tekrar denenebilir olup olmadığını belirler.
    """

    error_text = str(error).lower()

    retryable_keywords = (
        "timeout",
        "timed out",
        "rate limit",
        "429",
        "500",
        "502",
        "503",
        "504",
        "temporarily unavailable",
        "temporary",
        "connection",
        "network",
        "server error",
        "service unavailable",
        "internal server error",
    )

    return any(
        keyword in error_text
        for keyword in retryable_keywords
    )

def record_failure(
    job_name,
    error,
    job_id=None,
    attempt=1,
    max_retries=DEFAULT_MAX_RETRIES
):

    state = load_recovery_state()

    job_key = get_job_key(
        job_name,
        job_id
    )

    now = datetime.now().isoformat()

    state[job_key] = {
        "job_name": job_name,
        "job_id": job_id,
        "status": "retrying",
        "attempt": attempt,
        "max_retries": max_retries,
        "retryable": is_retryable_error(error),
        "last_error": str(error),
        "last_error_at": now,
    }

    save_recovery_state(state)

    return state[job_key]

def run_with_retry(
    function,
    *args,
    job_name="unknown",
    job_id=None,
    max_retries=DEFAULT_MAX_RETRIES,
    retry_delay=DEFAULT_RETRY_DELAY,
    **kwargs
):

    previous_state = get_job_state(
        job_name,
        job_id
    )

    if previous_state:
        start_attempt = (
            previous_state.get("attempt", 0)
            + 1
        )
    else:
        start_attempt = 1

    for attempt in range(
        start_attempt,
        max_retries + 1
    ):

        print()
        print(
            "========================================"
        )
        print(
            "🛡️ CREATOR ENGINE — RECOVERY"
        )
        print(
            "========================================"
        )

        print(
            f"🔧 İş      : {job_name}"
        )

        if job_id is not None:
            print(
                f"🆔 ID      : {job_id}"
            )

        print(
            f"🔄 Deneme  : "
            f"{attempt}/{max_retries}"
        )

        try:

            result = function(
                *args,
                **kwargs
            )

            clear_failure(
                job_name,
                job_id
            )

            print(
                "✅ İşlem başarıyla tamamlandı."
            )

            return {
                "status": "success",
                "result": result,
                "attempt": attempt,
            }

        except Exception as error:

            retryable = is_retryable_error(
                error
            )

            print(
                f"❌ Hata: {error}"
            )

            record_failure(
                job_name=job_name,
                error=error,
                job_id=job_id,
                attempt=attempt,
                max_retries=max_retries
            )

            if not retryable:

                print(
                    "🛑 Hata tekrar denenebilir "
                    "türden değil."
                )

                return {
                    "status": "failed",
                    "retryable": False,
                    "error": str(error),
                    "attempt": attempt,
                }

            if attempt >= max_retries:

                print(
                    "🛑 Maksimum retry sayısına "
                    "ulaşıldı."
                )

                state = load_recovery_state()

                job_key = get_job_key(
                    job_name,
                    job_id
                )

                if job_key in state:
                    state[job_key][
                        "status"
                    ] = "failed"

                    save_recovery_state(
                        state
                    )

                return {
                    "status": "failed",
                    "retryable": True,
                    "error": str(error),
                    "attempt": attempt,
                }

            print(
                f"⏳ {retry_delay} saniye "
                "sonra tekrar denenecek..."
            )

            time.sleep(
                retry_delay
            )

    return {
        "status": "failed",
        "error": "Retry döngüsü beklenmedik şekilde sona erdi."
    }

def parse_duration(duration):

    import re

    match = re.fullmatch(
        r"PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?",
        duration
    )

    if not match:
        return 0

    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    seconds = int(match.group(3) or 0)

    return (
        hours * 3600
        + minutes * 60
        + seconds
    )

    video_file,
    title,
    description="",
    tags=None,
    publish_at=None,

    video_file = Path(video_file)

    if not video_file.exists():
        raise FileNotFoundError(
            f"Video bulunamadı: {video_file}"
        )

    if tags is None:
        tags = []

    print()
    print("========================================")
    print("📤 YOUTUBE VIDEO YÜKLEME")
    print("========================================")
    print()
    print(f"🎬 Video : {video_file}")
    print(f"📝 Başlık: {title}")

    if publish_at:
        print(
            f"📅 Yayın zamanı: {publish_at}"
        )

    body = {
        "snippet": {
            "title": title,
            "description": description,
            "tags": tags,
            "categoryId": "22",
        },
        "status": {
            "privacyStatus": "private",
        },
    }

    if publish_at:
        body["status"]["publishAt"] = publish_at

    media = MediaFileUpload(
        str(video_file),
        mimetype="video/mp4",
        resumable=True,
    )

    print()
    print("📤 YouTube'a yükleme başlıyor...")

    request = youtube.videos().insert(
        part="snippet,status",
        body=body,
        media_body=media,
    )

    response = None

    while response is None:

        status, response = request.next_chunk()

        if status:
            progress = int(
                status.progress() * 100
            )

            print(
                f"📤 Yükleme: %{progress}"
            )

    video_id = response.get("id")

        raise RuntimeError(
            "YouTube video ID döndürmedi."

    print("✅ YOUTUBE YÜKLEME BAŞARILI")

    print()
    print(f"🆔 Video ID: {video_id}")
    print(
        f"🔗 https://www.youtube.com/watch?v={video_id}"
    )
    print()

    return video_id
