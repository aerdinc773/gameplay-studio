def row_to_dict(row):

    if isinstance(row, dict):

        return row

    return dict(row)


def is_short(video):

    video = row_to_dict(video)

    duration = (
        video.get("duration")
        or 0
    )

    topic = (
        video.get("topic")
        or ""
    ).lower()


    # --------------------------------------------------------
    # YouTube sync ile aynı Shorts mantığı
    # 180 saniyeden uzun videolar Shorts değildir.
    # --------------------------------------------------------

    if duration > 180:

        return False


    # --------------------------------------------------------
    # Canlı yayınları çıkar
    # --------------------------------------------------------

    live_keywords = (
        "kanalı canlı yayında",
        "canlı yayında",
        "canlı yayın",
        "live stream",
        "livestream",
    )


    for keyword in live_keywords:

        if keyword in topic:

            return False


    return True

def generate_image(prompt, output_path):
    result = client.images.generate(

        prompt=prompt,
        size="1024x1024",
        quality="low",
    )

    image_data = result.data[0].b64_json

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    with open(output_path, "wb") as f:
        f.write(base64.b64decode(image_data))

def get_all_video_ids(youtube):

    response = youtube.channels().list(
        part="contentDetails",
        mine=True
    ).execute()

    channels = response.get("items", [])

    if not channels:
        return []

    uploads_playlist = (
        channels[0]
        ["contentDetails"]
        ["relatedPlaylists"]
        ["uploads"]
    )

    video_ids = []
    page_token = None

    while True:

        response = youtube.playlistItems().list(
            part="contentDetails",
            playlistId=uploads_playlist,
            maxResults=50,
            pageToken=page_token
        ).execute()

        for item in response.get("items", []):

            video_ids.append(
                item["contentDetails"]["videoId"]
            )

        page_token = response.get(
            "nextPageToken"
        )

        if not page_token:
            break

    return video_ids

def parse_duration(duration):

    import re

    match = re.fullmatch(
        r"PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?",
        duration
    )

    if not match:
        return 0

    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    seconds = int(match.group(3) or 0)

    return (
        hours * 3600
        + minutes * 60
        + seconds
    )

def get_shorts(videos):

    shorts = []

    for video in videos:

        duration = parse_duration(
            video["contentDetails"]["duration"]
        )

        # Önceden gördüğümüz kanal yapısına göre

        #
        # 3 dakikaya kadar Shorts olabileceği için
        # sadece süreye göre 60 saniye sınırı koymuyoruz.

        title = video["snippet"].get(
            "title",
            ""
        )

        # Canlı yayın
        if title.lower().find(
            "kanalı canlı yayında"
        ) >= 0:

            continue

        # 3 dakikadan uzun içerikler
        if duration > 180:

            continue

        shorts.append(video)

    return shorts

def get_video_analytics(
    analytics,
    video_id
):

    try:

        response = analytics.reports().query(
            ids="channel==MINE",
            startDate="2020-01-01",
            endDate="2030-01-01",
            metrics=(
                "views,"
                "likes,"
                "comments,"
                "estimatedMinutesWatched,"
                "averageViewDuration,"
                "averageViewPercentage"
            ),
            filters=f"video=={video_id}"
        ).execute()

        rows = response.get(
            "rows",

        )

        if not rows:

            "views": int(row[0]),

            "comments": int(row[2]),
            "estimated_minutes": float(row[3]),
            "average_watch_time": float(row[4]),

        print(
            f"⚠️ Analytics alınamadı: "

        )

        print(
            f"   {error}"
        )

        return None

def get_existing_video_numbers():

            youtube_id
        FROM videos
        WHERE youtube_id IS NOT NULL

    return {
        youtube_id: video_number
        for video_number, youtube_id
        in rows
        if youtube_id
    }


def get_next_video_number():

    db = sqlite3.connect(DB_FILE)

    row = db.execute(
        """
        SELECT COALESCE(
            MAX(video_number),
            0
        )
        FROM videos
        """
    ).fetchone()

    db.close()

    return int(row[0]) + 1

    try:

        with open(
            STATE_FILE,
            "r",

        ) as file:

            return json.load(file)

    except Exception as error:

        print(
            f"⚠️ Analiz durumu okunamadı: {error}"
        )

        return {
            "last_analysis_at": None,
            "last_analyzed_video_count": 0
        }


def save_analysis_state(
    last_analysis_at,
    last_analyzed_video_count
):

    STATE_FILE.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    state = {
        "last_analysis_at": last_analysis_at,
        "last_analyzed_video_count":
            last_analyzed_video_count
    }

    with open(
        STATE_FILE,
        "w",
        encoding="utf-8"
    ) as file:

        json.dump(
            state,
            file,
            ensure_ascii=False,
            indent=4
        )

def get_new_video_count(
    total_video_count,
    last_analyzed_video_count
):

    return max(
        0,
        total_video_count
        - last_analyzed_video_count
    )


def get_days_since_analysis(
    last_analysis_at
):

    if not last_analysis_at:
        return None

    try:

        last_date = datetime.fromisoformat(
            last_analysis_at
        )

        elapsed = (
            datetime.now()
            - last_date
        )

        return (
            elapsed.total_seconds()
            / 86400
        )

    except Exception:

        return None

def format_srt_time(seconds):

    milliseconds = int(round(seconds * 1000))

    hours = milliseconds // 3_600_000
    milliseconds %= 3_600_000

    minutes = milliseconds // 60_000
    milliseconds %= 60_000

    seconds = milliseconds // 1000
    milliseconds %= 1000

    return (
        f"{hours:02d}:"
        f"{minutes:02d}:"
        f"{seconds:02d},"
        f"{milliseconds:03d}"
    )

def create_subtitle(
    transcript_file,
    audio_file,
    output_file,
):

    transcript_file = Path(transcript_file)
    audio_file = Path(audio_file)
    output_file = Path(output_file)

    if not transcript_file.exists():
        raise FileNotFoundError(
            f"Transkript bulunamadı: {transcript_file}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    text = transcript_file.read_text(
        encoding="utf-8"
    ).strip()

    if not text:
        raise ValueError(
            "Transkript boş."
        )

    duration = get_audio_duration(
        audio_file
    )

    print()
    print("=== AI ALTYAZI MOTORU ===")
    print(
        f"🎙️ Ses süresi: {duration:.2f} saniye"
    )

    print(
        "🤖 AI doğal altyazı bloklarını oluşturuyor..."
    )

    captions = create_ai_captions(
        text
    )

    print(
        f"📝 {len(captions)} altyazı bloğu oluşturuldu."
    )

    # --------------------------------------------------------
    # KARAKTER TOPLAMI
    # --------------------------------------------------------

    total_characters = sum(
        len(caption)
        for caption in captions
    )

    if total_characters <= 0:
        raise ValueError(
            "Altyazı karakter sayısı geçersiz."
        )

    # --------------------------------------------------------
    # ZAMANLAMA
    # --------------------------------------------------------

    current_time = 0.0

    srt_blocks = []

    for index, caption in enumerate(
        captions,
        start=1
    ):

        ratio = (
            len(caption)
            / total_characters
        )

        segment_duration = (
            duration * ratio
        )

        start_time = current_time

        if index == len(captions):

            end_time = duration

        else:

            end_time = (
                current_time
                + segment_duration
            )

        srt_blocks.append(
            f"{index}\n"
            f"{format_srt_time(start_time)} --> "
            f"{format_srt_time(end_time)}\n"
            f"{caption}\n"
        )

        current_time = end_time

    # --------------------------------------------------------
    # KAYDET
    # --------------------------------------------------------

    output_file.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    output_file.write_text(
        "\n".join(srt_blocks),
        encoding="utf-8"
    )

    print()
    print("=== ALTYAZI BLOKLARI ===")

    for index, caption in enumerate(
        captions,
        start=1
    ):
        print(
            f"{index:02d}: {caption}"
        )

    print()
    print("✅ AI ALTYAZI OLUŞTURULDU")
    print(f"Dosya: {output_file}")

    return output_file

def create_video(
    image_folder,
    audio_file,
    output_file,
):

    image_folder = Path(image_folder)
    audio_file = Path(audio_file)
    output_file = Path(output_file)

    # --------------------------------------------------------
    # DOSYA KONTROLLERİ
    # --------------------------------------------------------

    if not image_folder.exists():
        raise FileNotFoundError(
            f"Görsel klasörü bulunamadı: {image_folder}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    images = sorted(
        image_folder.glob("*.png")
    )

    if not images:
        raise FileNotFoundError(
            f"Görsel bulunamadı: {image_folder}"
        )

    # --------------------------------------------------------
    # SES SÜRESİNİ ÖLÇ
    # --------------------------------------------------------

    audio_duration = get_audio_duration(
        audio_file
    )

    # --------------------------------------------------------
    # GÖRSELLERİN SÜRESİNİ HESAPLA
    # --------------------------------------------------------

    image_duration = (
        audio_duration / len(images)
    )

    print()
    print("=== VİDEO OLUŞTURMA ===")
    print(
        f"🎨 Görsel sayısı: {len(images)}"
    )
    print(
        f"🎙️ Ses süresi: {audio_duration:.2f} saniye"
    )
    print(
        f"⏱️ Görsel başına süre: "
        f"{image_duration:.2f} saniye"
    )
    print(
        "🎬 Video oluşturuluyor..."
    )

    # --------------------------------------------------------
    # GEÇİCİ GÖRSEL LİSTESİ
    # --------------------------------------------------------

    concat_file = (
        image_folder.parent
        / "image_list.txt"
    )

    with open(
        concat_file,
        "w",
        encoding="utf-8"
    ) as file:

        for image in images:

            file.write(
                f"file '{image.resolve()}'\n"
            )

            file.write(
                f"duration {image_duration:.6f}\n"
            )

        # FFmpeg concat demuxer'da son görselin
        # duration değerinin uygulanması için
        # son dosyayı tekrar ekliyoruz.

        file.write(
            f"file '{images[-1].resolve()}'\n"
        )

    # --------------------------------------------------------
    # FFMPEG
    # --------------------------------------------------------

    command = [
        "ffmpeg",
        "-y",

        "-f",
        "concat",

        "-safe",
        "0",

        "-i",
        str(concat_file),

        "-i",
        str(audio_file),

        # ----------------------------------------------------
        # SHORTS FORMAT
        # ----------------------------------------------------

        "-vf",
        (
            "scale=1080:1920:"
            "force_original_aspect_ratio=increase,"
            "crop=1080:1920"
        ),

        "-r",
        "25",

        # ----------------------------------------------------
        # VİDEO
        # ----------------------------------------------------

        "-c:v",
        "libx264",

        "-preset",
        "medium",

        "-crf",
        "20",

        "-pix_fmt",
        "yuv420p",

        # ----------------------------------------------------
        # SES
        # ----------------------------------------------------

        "-c:a",
        "aac",

        "-b:a",
        "128k",

        # ----------------------------------------------------
        # SESİ KESME
        # ----------------------------------------------------

        "-t",
        f"{audio_duration:.3f}",

        str(output_file),
    ]

    # --------------------------------------------------------
    # ÇALIŞTIR
    # --------------------------------------------------------

    subprocess.run(
        command,
        check=True
    )

    # --------------------------------------------------------
    # GEÇİCİ DOSYAYI SİL
    # --------------------------------------------------------

    if concat_file.exists():
        concat_file.unlink()

    # --------------------------------------------------------
    # SONUÇ
    # --------------------------------------------------------

    if not output_file.exists():
        raise RuntimeError(
            "FFmpeg çalıştı ancak video oluşmadı."
        )

    print()
    print("========================================")
    print("✅ VIDEO OLUŞTURULDU")
    print("========================================")
    print()
    print(
        f"Dosya: {output_file}"
    )

    return output_file

from google.oauth2.credentials import Credentials

# CREATOR ENGINE — YOUTUBE OAUTH

BASE_DIR = Path(__file__).resolve().parents[2]

# Google Cloud'dan indirdiğin OAuth JSON dosyası
CLIENT_SECRET_FILE = (
    BASE_DIR
    / "client_secret_150956391796-vpb251vj5d9dib3qccgpjqjmmnjmv4p9.apps.googleusercontent.com.json"
)

# OAuth sonrası oluşturulacak token
TOKEN_FILE = (
    BASE_DIR
    / "data"
    / "youtube_token.json"
)


# ============================================================
# YOUTUBE İZİNLERİ
# ============================================================

SCOPES = [
    "https://www.googleapis.com/auth/youtube.readonly",
    "https://www.googleapis.com/auth/yt-analytics.readonly",
]


# ============================================================
# OAUTH BAĞLANTISI
# ============================================================

def authenticate_youtube():

    print()
    print("========================================")
    print("🔐 CREATOR ENGINE — YOUTUBE BAĞLANTISI")
    print("========================================")
    print()

    if not CLIENT_SECRET_FILE.exists():

        print("❌ OAuth JSON dosyası bulunamadı.")
        print()
        print(
            f"Aranan dosya:"
        )
        print(
            CLIENT_SECRET_FILE
        )

        return None

    credentials = None

    # --------------------------------------------------------
    # Daha önce alınmış token varsa kullan
    # --------------------------------------------------------

    if TOKEN_FILE.exists():

        try:

            credentials = Credentials.from_authorized_user_file(
                TOKEN_FILE,
                SCOPES,
            )

            print("✅ Mevcut YouTube yetkilendirmesi bulundu.")

        except Exception as error:

            print()
            print(
                f"⚠️ Mevcut token okunamadı: {error}"
            )

            credentials = None

    # --------------------------------------------------------
    # Yeni OAuth bağlantısı
    # --------------------------------------------------------

    if credentials is None or not credentials.valid:

        print()
        print("🌐 Google OAuth başlatılıyor...")
        print()
        print(
            "Tarayıcı açılacak."
        )
        print(
            "33 Shorts bulunan YouTube kanalının"
        )
        print(
            "Google hesabıyla giriş yap."
        )
        print()

        try:

            flow = InstalledAppFlow.from_client_secrets_file(
                CLIENT_SECRET_FILE,
                SCOPES,
            )

            credentials = flow.run_local_server(
                port=0
            )

        except Exception as error:

            print()
            print(
                f"❌ OAuth bağlantısı başarısız: {error}"
            )

            return None

        # ----------------------------------------------------
        # Token kaydet
        # ----------------------------------------------------

        TOKEN_FILE.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        TOKEN_FILE.write_text(
            credentials.to_json(),
            encoding="utf-8"
        )

        print()
        print("✅ YouTube yetkilendirmesi tamamlandı.")
        print()
        print(
            f"🔐 Token kaydedildi:"
        )
        print(
            TOKEN_FILE
        )

    print()
    print("========================================")
    print("✅ YOUTUBE BAĞLANTISI HAZIR")
    print("========================================")
    print()

    return credentials

if __name__ == "__main__":
