def analyze_videos():

    initialize_database()

    videos = get_all_videos()

    # ========================================================
    # VERİ YOK
    # ========================================================

    if not videos:

        return {
            "total_videos": 0,
            "average_duration": 0,
            "average_images": 0,
            "total_views": 0,
            "total_likes": 0,
            "total_comments": 0,
            "average_views": 0,
            "average_likes": 0,
            "average_comments": 0,
            "top_topics": [],
            "best_video": None,
            "top_videos": [],
            "worst_videos": [],
            "duration_performance": [],
            "engagement_rate": 0,
            "average_watch_time": 0,
            "average_watch_percentage": 0,
            "above_average_videos": 0,
            "performance_available": False,
        }

    # ========================================================
    # TEMEL VERİLER
    # ========================================================

    total_videos = len(videos)

    durations = [
        video["duration"]
        for video in videos
        if video["duration"] is not None
        and video["duration"] > 0
    ]

    image_counts = [
        video["image_count"]
        for video in videos
        if video["image_count"] is not None
        and video["image_count"] > 0
    ]

    total_views = sum(
        video["views"] or 0
        for video in videos
    )

    total_likes = sum(
        video["likes"] or 0
        for video in videos
    )

    total_comments = sum(
        video["comments"] or 0
        for video in videos
    )

    # ========================================================
    # ORTALAMALAR
    # ========================================================

    average_duration = (
        sum(durations) / len(durations)
        if durations
        else 0
    )

    average_images = (
        sum(image_counts) / len(image_counts)
        if image_counts
        else 0
    )

    average_views = (
        total_views / total_videos
        if total_videos
        else 0
    )

    average_likes = (
        total_likes / total_videos
        if total_videos
        else 0
    )

    average_comments = (
        total_comments / total_videos
        if total_videos
        else 0
    )

    # ========================================================
    # PERFORMANS VERİSİ
    # ========================================================

    performance_videos = [
        video
        for video in videos
        if (video["views"] or 0) > 0
    ]

    performance_available = bool(
        performance_videos
    )

    # ========================================================
    # WATCH TIME
    # ========================================================

    watch_time_values = [
        video["average_watch_time"]
        for video in videos
        if video["average_watch_time"] is not None
        and video["average_watch_time"] > 0
    ]

    watch_percentage_values = [
        video["watch_percentage"]
        for video in videos
        if video["watch_percentage"] is not None
        and video["watch_percentage"] > 0
    ]

    average_watch_time = (
        sum(watch_time_values)
        / len(watch_time_values)
        if watch_time_values
        else 0
    )

    average_watch_percentage = (
        sum(watch_percentage_values)
        / len(watch_percentage_values)
        if watch_percentage_values
        else 0
    )

    # ========================================================
    # ETKİLEŞİM ORANI
    # ========================================================

    engagement_rate = (
        (
            total_likes
            + total_comments
        )
        / total_views
        * 100
        if total_views > 0
        else 0
    )

    # ========================================================
    # KONULAR
    # ========================================================

    topics = [
        video["topic"]
        for video in videos
        if video["topic"]
    ]

    topic_counter = Counter(topics)

    top_topics = [
        {
            "topic": topic,
            "count": count,
        }
        for topic, count
        in topic_counter.most_common(10)
    ]

    # ========================================================
    # EN İYİ VİDEOLAR
    # ========================================================

    sorted_videos = sorted(
        performance_videos,
        key=lambda video: (
            video["views"] or 0
        ),
        reverse=True,
    )

    top_videos = [
        {
            "video_number":
                video["video_number"],

            "topic":
                video["topic"],

            "views":
                video["views"] or 0,

            "likes":
                video["likes"] or 0,

            "comments":
                video["comments"] or 0,

            "duration":
                video["duration"] or 0,

            "watch_percentage":
                video["watch_percentage"] or 0,
        }
        for video in sorted_videos[:10]
    ]

    # ========================================================
    # EN KÖTÜ VİDEOLAR
    # ========================================================

    worst_sorted = sorted(
        performance_videos,
        key=lambda video: (
            video["views"] or 0
        ),
    )

    worst_videos = [
        {
            "video_number":
                video["video_number"],

            "topic":
                video["topic"],

            "views":
                video["views"] or 0,

            "duration":
                video["duration"] or 0,
        }
        for video in worst_sorted[:10]
    ]

    # ========================================================
    # EN BAŞARILI VİDEO
    # ========================================================

    best_video = None

    if sorted_videos:

        best = sorted_videos[0]

        best_video = {
            "video_number":
                best["video_number"],

            "topic":
                best["topic"],

            "views":
                best["views"] or 0,

            "likes":
                best["likes"] or 0,

            "comments":
                best["comments"] or 0,

            "duration":
                best["duration"] or 0,

            "watch_percentage":
                best["watch_percentage"] or 0,
        }

    # ========================================================
    # ORTALAMA ÜZERİ VİDEOLAR
    # ========================================================

    above_average_videos = len([
        video
        for video in performance_videos
        if (video["views"] or 0)
        > average_views
    ])

    # ========================================================
    # SÜRE PERFORMANSI
    # ========================================================

    duration_groups = {
        "0-20 sn": [],
        "21-30 sn": [],
        "31-45 sn": [],
        "46-60 sn": [],
        "61+ sn": [],
    }

    for video in performance_videos:

        duration = video["duration"] or 0
        views = video["views"] or 0

        if duration <= 20:

            duration_groups[
                "0-20 sn"
            ].append(views)

        elif duration <= 30:

            duration_groups[
                "21-30 sn"
            ].append(views)

        elif duration <= 45:

            duration_groups[
                "31-45 sn"
            ].append(views)

        elif duration <= 60:

            duration_groups[
                "46-60 sn"
            ].append(views)

        else:

            duration_groups[
                "61+ sn"
            ].append(views)

    duration_performance = []

    for group, values in duration_groups.items():

        if values:

            duration_performance.append({
                "range": group,
                "video_count": len(values),
                "average_views": round(
                    sum(values) / len(values),
                    2
                ),
                "total_views": sum(values),
            })

    # ========================================================
    # SONUÇ
    # ========================================================

    return {

        "total_videos":
            total_videos,

        "average_duration":
            round(
                average_duration,
                2
            ),

        "average_images":
            round(
                average_images,
                2
            ),

        "total_views":
            total_views,

        "total_likes":
            total_likes,

        "total_comments":
            total_comments,

        "average_views":
            round(
                average_views,
                2
            ),

        "average_likes":
            round(
                average_likes,
                2
            ),

        "average_comments":
            round(
                average_comments,
                2
            ),

        "top_topics":
            top_topics,

        "best_video":
            best_video,

        "top_videos":
            top_videos,

        "worst_videos":
            worst_videos,

        "duration_performance":
            duration_performance,

        "engagement_rate":
            round(
                engagement_rate,
                2
            ),

        "average_watch_time":
            round(
                average_watch_time,
                2
            ),

        "average_watch_percentage":
            round(
                average_watch_percentage,
                2
            ),

        "above_average_videos":
            above_average_videos,

        "performance_available":
            performance_available,
    }

def show_schema():

    print()
    print("========================================")
    print("🧠 CREATOR ENGINE — DATABASE SCHEMA")
    print("========================================")
    print()

    print(f"📁 Veritabanı: {DB_FILE}")
    print()

    db = sqlite3.connect(DB_FILE)
    cursor = db.cursor()

    tables = cursor.execute(
        """
        SELECT name
        FROM sqlite_master
        WHERE type = 'table'
        ORDER BY name
        """
    ).fetchall()

    print("=== TABLOLAR ===")
    print()

    for table in tables:

        table_name = table[0]

        print(f"📦 {table_name}")

        columns = cursor.execute(
            f'PRAGMA table_info("{table_name}")'
        ).fetchall()

        for column in columns:

            print(
                f"   {column[1]} | "
                f"{column[2]} | "
                f"PK={column[5]}"
            )

        print()

    db.close()

    print("========================================")
    print("✅ ŞEMA OKUMA TAMAMLANDI")
    print("========================================")
    print()

def parse_duration(duration):

    import re

    match = re.fullmatch(
        r"PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?",
        duration
    )

    if not match:
        return 0

    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    seconds = int(match.group(3) or 0)

    return (
        hours * 3600
        + minutes * 60
        + seconds
    )

def format_srt_time(seconds):

    milliseconds = int(round(seconds * 1000))

    hours = milliseconds // 3_600_000
    milliseconds %= 3_600_000

    minutes = milliseconds // 60_000
    milliseconds %= 60_000

    seconds = milliseconds // 1000
    milliseconds %= 1000

    return (
        f"{hours:02d}:"
        f"{minutes:02d}:"
        f"{seconds:02d},"
        f"{milliseconds:03d}"
    )

def create_subtitle(
    transcript_file,
    audio_file,
    output_file,
):

    transcript_file = Path(transcript_file)
    audio_file = Path(audio_file)
    output_file = Path(output_file)

    if not transcript_file.exists():
        raise FileNotFoundError(
            f"Transkript bulunamadı: {transcript_file}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    text = transcript_file.read_text(
        encoding="utf-8"
    ).strip()

    if not text:
        raise ValueError(
            "Transkript boş."
        )

    duration = get_audio_duration(
        audio_file
    )

    print()
    print("=== AI ALTYAZI MOTORU ===")
    print(
        f"🎙️ Ses süresi: {duration:.2f} saniye"
    )

    print(
        "🤖 AI doğal altyazı bloklarını oluşturuyor..."
    )

    captions = create_ai_captions(
        text
    )

    print(
        f"📝 {len(captions)} altyazı bloğu oluşturuldu."
    )

    # --------------------------------------------------------
    # KARAKTER TOPLAMI
    # --------------------------------------------------------

    total_characters = sum(
        len(caption)
        for caption in captions
    )

    if total_characters <= 0:
        raise ValueError(
            "Altyazı karakter sayısı geçersiz."
        )

    # --------------------------------------------------------
    # ZAMANLAMA
    # --------------------------------------------------------

    current_time = 0.0

    srt_blocks = []

    for index, caption in enumerate(
        captions,
        start=1
    ):

        ratio = (
            len(caption)
            / total_characters
        )

        segment_duration = (
            duration * ratio
        )

        start_time = current_time

        if index == len(captions):

            end_time = duration

        else:

            end_time = (
                current_time
                + segment_duration
            )

        srt_blocks.append(
            f"{index}\n"
            f"{format_srt_time(start_time)} --> "
            f"{format_srt_time(end_time)}\n"
            f"{caption}\n"
        )

        current_time = end_time

    # --------------------------------------------------------
    # KAYDET
    # --------------------------------------------------------

    output_file.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    output_file.write_text(
        "\n".join(srt_blocks),
        encoding="utf-8"
    )

    print()
    print("=== ALTYAZI BLOKLARI ===")

    for index, caption in enumerate(
        captions,
        start=1
    ):
        print(
            f"{index:02d}: {caption}"
        )

    print()
    print("✅ AI ALTYAZI OLUŞTURULDU")
    print(f"Dosya: {output_file}")

    return output_file

        image_folder.parent

        concat_file,

    ) as file:

        for image in images:

            file.write(
                f"file '{image.resolve()}'\n"
            )

            )

        # FFmpeg concat demuxer'da son görselin

def authenticate_youtube():

    print()
    print("========================================")
    print("🔐 CREATOR ENGINE — YOUTUBE BAĞLANTISI")
    print("========================================")
    print()

    if not CLIENT_SECRET_FILE.exists():

        print("❌ OAuth JSON dosyası bulunamadı.")
        print()
        print(
            f"Aranan dosya:"
        )
        print(
            CLIENT_SECRET_FILE
        )

        return None

    credentials = None

    # --------------------------------------------------------
    # Daha önce alınmış token varsa kullan
    # --------------------------------------------------------

    if TOKEN_FILE.exists():

        try:

            credentials = Credentials.from_authorized_user_file(
                TOKEN_FILE,
                SCOPES,
            )

            print("✅ Mevcut YouTube yetkilendirmesi bulundu.")

        except Exception as error:

            print()
            print(
                f"⚠️ Mevcut token okunamadı: {error}"
            )

            credentials = None

    # --------------------------------------------------------
    # Yeni OAuth bağlantısı
    # --------------------------------------------------------

    if credentials is None or not credentials.valid:

        print()
        print("🌐 Google OAuth başlatılıyor...")
        print()
        print(
            "Tarayıcı açılacak."
        )
        print(
            "33 Shorts bulunan YouTube kanalının"
        )
        print(
            "Google hesabıyla giriş yap."
        )
        print()

        try:

            flow = InstalledAppFlow.from_client_secrets_file(
                CLIENT_SECRET_FILE,
                SCOPES,
            )

            credentials = flow.run_local_server(
                port=0
            )

        except Exception as error:

            print()
            print(
                f"❌ OAuth bağlantısı başarısız: {error}"
            )

            return None

        # ----------------------------------------------------
        # Token kaydet
        # ----------------------------------------------------

        TOKEN_FILE.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        TOKEN_FILE.write_text(
            credentials.to_json(),
            encoding="utf-8"
        )

        print()
        print("✅ YouTube yetkilendirmesi tamamlandı.")
        print()
        print(
            f"🔐 Token kaydedildi:"
        )
        print(
            TOKEN_FILE
        )

    print()
    print("========================================")
    print("✅ YOUTUBE BAĞLANTISI HAZIR")
    print("========================================")
    print()

    return credentials
